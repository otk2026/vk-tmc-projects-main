```
=IF(TRIM(I17&"")="","",IF(AND(Y17="",Z17=""),IF(OR(ISNUMBER(SEARCH("не соотв",I17)),ISNUMBER(SEARCH("несоотв",I17))),"Не соответствует",IF(ISNUMBER(SEARCH("соответ",I17)),"Соответствует","Не соответствует")),IF(NOT(ISNUMBER(I17)),"Не соответствует",IF(AND(OR(Y17="",I17>=Y17-0.000001),OR(Z17="",I17<=Z17+0.000001)),"Соответствует","Не соответствует"))))
```
